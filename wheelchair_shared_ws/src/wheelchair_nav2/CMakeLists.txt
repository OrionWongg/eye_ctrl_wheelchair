cmake_minimum_required(VERSION 3.8)
project(wheelchair_nav2)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp nav2_util nav2_core nav2_costmap_2d dwb_core dwb_msgs nav_2d_msgs nav_msgs geometry_msgs pluginlib wheelchair_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
add_library(fixed_action_generator SHARED src/fixed_action_generator.cpp)
target_include_directories(fixed_action_generator PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(fixed_action_generator rclcpp nav2_util dwb_core dwb_msgs nav_2d_msgs pluginlib)
pluginlib_export_plugin_description_file(dwb_core plugins.xml)
add_executable(discrete_path_planner src/discrete_path_planner.cpp)
ament_target_dependencies(discrete_path_planner rclcpp nav2_util nav2_costmap_2d dwb_core dwb_msgs nav_2d_msgs nav_msgs geometry_msgs pluginlib wheelchair_interfaces)
add_executable(smac_lattice_interface src/smac_lattice_interface.cpp)
ament_target_dependencies(smac_lattice_interface rclcpp nav2_util nav2_core nav2_costmap_2d nav_msgs geometry_msgs pluginlib wheelchair_interfaces)
add_executable(wheelchair_costmap src/costmap_node.cpp)
ament_target_dependencies(wheelchair_costmap rclcpp nav2_costmap_2d)
install(TARGETS fixed_action_generator discrete_path_planner wheelchair_costmap smac_lattice_interface ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(dwb_core nav2_util)
install(FILES LICENSE.costmap DESTINATION share/${PROJECT_NAME})
ament_package()
