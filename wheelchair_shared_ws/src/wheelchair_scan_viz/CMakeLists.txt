cmake_minimum_required(VERSION 3.16)
project(wheelchair_scan_viz)
find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Quick Qml Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 17)
set(GAZEBO_RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/runtime" CACHE PATH "Local Gazebo Fortress runtime prefix")
file(GLOB IGNITION_INCLUDES CONFIGURE_DEPENDS "${GAZEBO_RUNTIME}/usr/include/ignition/*")
file(GLOB SDF_INCLUDES "${GAZEBO_RUNTIME}/usr/include/sdformat-*")
add_library(ScanPoints SHARED src/ScanPoints.cc src/ScanPoints.qrc)
target_include_directories(ScanPoints PRIVATE ${IGNITION_INCLUDES} ${SDF_INCLUDES} "${GAZEBO_RUNTIME}/usr/include" "${GAZEBO_RUNTIME}/usr/include/aarch64-linux-gnu/qt5" "${GAZEBO_RUNTIME}/usr/include/aarch64-linux-gnu/qt5/QtQuickControls2")
target_link_libraries(ScanPoints Qt5::Core Qt5::Quick Qt5::Qml Qt5::Widgets)
foreach(component gui6 rendering6 transport11 msgs8 math6 common4 gazebo6-gui gazebo6)
  find_library(LIB_${component} NAMES ignition-${component} PATHS "${GAZEBO_RUNTIME}/usr/lib/aarch64-linux-gnu" REQUIRED NO_DEFAULT_PATH)
  target_link_libraries(ScanPoints ${LIB_${component}})
endforeach()
find_library(PROTOBUF NAMES protobuf libprotobuf.so.23 REQUIRED)
target_link_libraries(ScanPoints ${PROTOBUF})
install(TARGETS ScanPoints LIBRARY DESTINATION lib)
ament_package()
